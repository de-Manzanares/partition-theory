\section{My Propositions}\label{sec:my-propositions}

\begin{proposition}
    The number of partitions of $n$ in which positive distinct parts $\lambda_1, \lambda_2,\dots,\lambda_m$ each appear
    at least once equals
    the number of partitions of $n-\sum_{i=1}^{m}\lambda_i$.
\end{proposition}

Proposition \autoref{prop:prop-ex-1-2} is directly inspired by Chapter 1 examples 1 and 2
from \emph{The Theory of Partitions} by George E. Andrews.

\begin{proposition}\label{prop:prop-ex-1-2}
    The following are equal:
    the number of partitions of $n$ in which multiples of $x$ are distinct,
    here denoted $p_\alpha(n)$;
    the number of partitions of $n$ in which no part is divisible by $2x$,
    here denoted $p_\beta(n)$;
    the number of partitions of $n$ in which no part appears more than $2x-1$ times,
    here denoted $p_\gamma(n)$.
\end{proposition}
\begin{proof}
    \begin{align*}
        \sum_{n\ge 0}p_\alpha(n)q^n &= \prod_{n\ge 1}\left(1+q^{xn}\right)\frac{1-q^{xn}}{1-q^n}\\[0.5em]
        &= \frac{(q^{2x};q^{2x})_\infty}{(q;q)_\infty}\\[0.5em]
        &= \sum_{n\ge 0}p_\beta(n)q^n.
    \end{align*}
    By Glashier's Theorem~(\autoref{thm:glashiers-thm})
    $p_\beta(n)$ = $p_\gamma(n)$.
\end{proof}

\section{Theorems}\label{sec:theorems}

\begin{theorem}[Glashier's Theorem]\label{thm:glashiers-thm}
    The number of partitions of $n$ in which no part appears more than $d$ times,
    here denoted $p_\alpha(n)$,
    equals
    the number of partitions of $n$ in which no part is divisible by $d+1$,
    here denoted $p_\beta(n)$.
\end{theorem}

\begin{proof}
    \begin{align*}
        \sum_{n\ge 0} p_\alpha(n)q^n &= \prod_{n\ge 1}\left(1+q^n+\cdots+q^{dn}\right)\\[0.5em]
        &= \prod_{n\ge 1}\frac{1-q^{(d+1)n}}{1-q^n}\\[0.5em]
        &= \frac{(1-q^{d+1})(1-q^{2(d+1)})\cdots}{(1-q)\cdots(1-q^{d+1})\cdots(1-q^{2(d+1)})\cdots}\\[0.5em]
        &= \prod_{n\ge 1,\,d+1\nmid n}\frac{1}{1-q^n}\\[0.5em]
        &= \sum_{n\ge 0} p_\beta(n)q^n.
    \end{align*}
\end{proof}
